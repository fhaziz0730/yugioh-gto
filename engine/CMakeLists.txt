cmake_minimum_required(VERSION 3.20)
project(yugioh_engine LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
include(FetchContent)
FetchContent_Declare(fmt GIT_REPOSITORY https://github.com/fmtlib/fmt.git GIT_TAG 10.2.1 GIT_SHALLOW TRUE)
FetchContent_Declare(dense GIT_REPOSITORY https://github.com/martinus/unordered_dense.git GIT_TAG v4.4.0 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(fmt dense)
find_package(Python COMPONENTS Interpreter Development REQUIRED)
execute_process(COMMAND ${Python_EXECUTABLE} -m pybind11 --cmakedir OUTPUT_VARIABLE pybind11_DIR OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
find_package(pybind11 CONFIG REQUIRED)
find_package(Lua 5.3 EXACT REQUIRED)
find_package(glog REQUIRED)
find_package(SQLiteCpp REQUIRED)
file(GLOB CORE_SOURCES "${CORE_DIR}/*.cpp")
add_library(ocgcore STATIC ${CORE_SOURCES})
target_include_directories(ocgcore PRIVATE ${LUA_INCLUDE_DIR})
target_link_libraries(ocgcore PUBLIC ${LUA_LIBRARIES})
target_compile_options(ocgcore PRIVATE -include cstring)
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/include/concurrentqueue")
file(CREATE_LINK /usr/include/concurrentqueue "${CMAKE_BINARY_DIR}/include/concurrentqueue/moodycamel" SYMBOLIC)
file(CREATE_LINK "${CORE_DIR}" "${CMAKE_BINARY_DIR}/include/ygopro-core" SYMBOLIC)
pybind11_add_module(ygopro_ygoenv NO_EXTRAS "${ENV_DIR}/ygoenv/ygoenv/ygopro/ygopro.cpp")
target_include_directories(ygopro_ygoenv PRIVATE "${ENV_DIR}/ygoenv" "${CMAKE_BINARY_DIR}/include" ${LUA_INCLUDE_DIR})
target_link_libraries(ygopro_ygoenv PRIVATE ocgcore fmt::fmt unordered_dense::unordered_dense glog::glog SQLiteCpp)
target_compile_options(ygopro_ygoenv PRIVATE -include cstring)
set_target_properties(ygopro_ygoenv PROPERTIES SUFFIX ".so" LIBRARY_OUTPUT_DIRECTORY "${ENV_DIR}/ygoenv/ygoenv/ygopro")
